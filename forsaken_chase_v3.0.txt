local function Clamp(v, mn, mx)
  v, mn, mx = tonumber(v) or 0, tonumber(mn) or 0, tonumber(mx) or 1
  if v < mn then return mn end
  if v > mx then return mx end
  return v
end
if not math.clamp then math.clamp = Clamp end
local function SafeDiv(a, b, fallback)
  a, b = tonumber(a), tonumber(b)
  if not b or b == 0 then return (fallback or 0) end
  if not a then return (fallback or 0) end
  return a / b
end

local function ValidateCamera()
  local ok, cam = pcall(function() return workspace.CurrentCamera end)
  if not ok or not cam then return nil end
  return cam
end

local function GetPlayerGui()
  local pg = nil
  pcall(function() pg = LocalPlayer:FindFirstChildOfClass("PlayerGui") end)
  if pg then return pg end
  pcall(function() pg = LocalPlayer:FindFirstChild("PlayerGui") end)
  if pg then return pg end
  pcall(function() pg = LocalPlayer:WaitForChild("PlayerGui", 10) end)
  return pg
end

local function IsSpectating()
  -- true when camera is not following our humanoid (spectate cams, death cams, freecam)
  local hum = nil
  pcall(function()
    local ch = LocalPlayer.Character
    if ch then hum = ch:FindFirstChildOfClass("Humanoid") end
  end)
  if not hum then return true end
  local cam = ValidateCamera()
  if not cam then return false end
  local subj = nil
  pcall(function() subj = cam.CameraSubject end)
  if subj and subj ~= hum then
    -- allow seated/vehicle edge: only count Humanoid/Part subjects that are not ours
    return true
  end
  -- some games set attribute / team named Spectator
  local attr = nil
  pcall(function() attr = LocalPlayer:GetAttribute("Spectating") end)
  if attr == true then return true end
  if LocalPlayer.Team and string.lower(LocalPlayer.Team.Name):find("spect") then return true end
  return false
end

local function PreloadIds(list)
  task.spawn(function()
    local cp = nil
    pcall(function() cp = game:GetService("ContentProvider") end)
    if not cp then return end
    local batch = {}
    for _, id in ipairs(list) do
      if id and id ~= "" and not PreloadedCache[id] then
        PreloadedCache[id] = true
        table.insert(batch, id)
      end
    end
    if #batch == 0 then return end
    -- PreloadAsync expects Instances; wrap ids in Sounds is not possible pre-create,
    -- so we use string-safe call guarded (works on most executors, skipped otherwise)
    pcall(function() cp:PreloadAsync(batch) end)
  end)
end

  for _, k in ipairs({"layer1","layer2","layer3","chase","idle","lastChase","lastChase1","lastChase2","lastLayer1","lastLayer2","lastLayer3","finalCustom"}) do

end

  CONFIG.PROFILE = tostring(name)

      local raw = nil

      if raw then writefile(ProfilePath(name), raw) end

  name = name or "Default"
  local ok = pcall(function()
    if typeof(isfile) == "function" and isfile(ProfilePath(name)) then
      local raw = readfile(ProfilePath(name))
      if raw and raw ~= "" then
        writefile(CONFIG_FILE, raw)
        LoadConfig()
        CONFIG.PROFILE = tostring(name)
        return true
      end
    end
    return false
  end)
  print("[Chase] Profile load: " .. tostring(name))
  return ok
end

-- Utility Functions
SafePcall = function(func, ...)
	local success, result = pcall(func, ...)
	if not success then
		warn("[Chase] Error: " .. tostring(result))
	end
	return success, result
end

-- Config Functions
function SaveConfig(silent)
	SafePcall(function()
		if typeof(writefile) ~= "function" then return end
		
		local data = {}
		for _, k in ipairs(SAVE_KEYS) do
			local v = CONFIG[k]
			if typeof(v) == "number" or typeof(v) == "string" or typeof(v) == "boolean" then
				data[k] = v
			elseif typeof(v) == "table" then
				local ok, _ = pcall(function() return #v end)
				if ok then
					local arr = {}
					for i, n in ipairs(v) do
						if typeof(n) == "number" or typeof(n) == "string" or typeof(n) == "boolean" then
							arr[i] = n
						end
					end
					data[k] = arr
				end
			end
		end
		
		data.SPEED_ENABLED_SAVED = false
		local js = HttpService:JSONEncode(data)
		writefile(CONFIG_FILE, js)
		if not silent then print("[Chase] Config saved") end
	end)
end

function LoadConfig()
	SafePcall(function()
		if typeof(isfile) ~= "function" or typeof(readfile) ~= "function" then return end
		if not isfile(CONFIG_FILE) then return end
		
		local raw = readfile(CONFIG_FILE)
		if not raw or raw == "" then return end
		
		local ok, data = pcall(function() return HttpService:JSONDecode(raw) end)
		if not ok or typeof(data) ~= "table" then return end
		
		for _, k in ipairs(SAVE_KEYS) do
			if data[k] ~= nil then
				if k == "LAYER_DISTANCES" or k == "LAST_CHANCE_LAYER_DISTANCES" then
					if typeof(data[k]) == "table" and #data[k] >= 3 then
						CONFIG[k] = {
							tonumber(data[k][1]) or CONFIG[k][1],
							tonumber(data[k][2]) or CONFIG[k][2],
							tonumber(data[k][3]) or CONFIG[k][3]
						}
					end
				else
					CONFIG[k] = data[k]
				end
			end
		end
		
		CONFIG.SPEED_ENABLED = false
		SafePcall(function()
			local idx = tonumber(CONFIG.CURRENT_THEME) or 1
			CONFIG.CURRENT_THEME = Clamp(math.floor(idx), 1, 8)
		end)
		print("[Chase] Config loaded")
	end)
end

local function EncodeName(name)
	return (name:gsub("[^%w%-%_%.%~]", function(c)
		return string.format("%%%02X", string.byte(c))
	end))
end

local function SafeAsset(path)
	if typeof(getcustomasset) == "function" then
		local ok, r = pcall(getcustomasset, path)
		if ok and r and r ~= "" then return r end
	end
	if typeof(getasset) == "function" then

	end
	return ""
end

-- Movement Systems
local function IsMovingBackward()
	local hum = GetHumanoid()
	local hrp = GetRootPart()
	if not hum or not hrp then return false end
	if hum.MoveDirection.Magnitude < 0.1 then return false end
	
	local st = hum:GetState()
	if st == Enum.HumanoidStateType.Jumping or st == Enum.HumanoidStateType.Freefall or st == Enum.HumanoidStateType.FallingDown then
		return false
	end
	
	local look = Vector3.new(hrp.CFrame.LookVector.X, 0, hrp.CFrame.LookVector.Z)
	if look.Magnitude < 0.01 then return false end
	look = look.Unit
	
	local mv = Vector3.new(hum.MoveDirection.X, 0, hum.MoveDirection.Z)
	if mv.Magnitude < 0.1 then return false end
	mv = mv.Unit
	
	return mv:Dot(look) < (CONFIG.BACKWARD_ANIM_THRESHOLD or -0.2)
end

local function UpdateBackwardAnim()
	local hum = GetHumanoid()
	if not hum then BackwardBlend = 0 return end
	local animator = hum:FindFirstChildOfClass("Animator")
	if not animator then BackwardBlend = 0 return end
	local _wantB = 0
	if CONFIG.BACKWARD_ANIM_ENABLED and IsMovingBackward() then _wantB = 1 end
	if BackwardBlend < _wantB then BackwardBlend = math.min(_wantB, BackwardBlend + 0.016 * 6)
	elseif BackwardBlend > _wantB then BackwardBlend = math.max(_wantB, BackwardBlend - 0.016 * 6) end
	
	local backward = BackwardBlend > 0.5
	local mult = CONFIG.BACKWARD_ANIM_SPEED or 1
	
	SafePcall(function()
		for _, tr in ipairs(animator:GetPlayingAnimationTracks()) do
			local n = string.lower(tr.Name or "")
			local isJumpFall = string.find(n, "jump", 1, true) or string.find(n, "fall", 1, true)
			local sp = tr.Speed
			
			if isJumpFall then
				if sp < -0.01 then tr:AdjustSpeed(math.abs(sp)) end
			else
				if not CONFIG.BACKWARD_ANIM_ENABLED then
					if sp < -0.01 then tr:AdjustSpeed(math.abs(sp)) end
				else
					if backward then
						if sp > 0.01 then tr:AdjustSpeed(-math.abs(sp) * mult) end
					else
						if sp < -0.01 then tr:AdjustSpeed(math.abs(sp)) end
					end
				end
			end
		end
	end)
end

local function UpdateAccel(dt)
	if not CONFIG.ACCEL_ENABLED then
		AccelCur = nil
		return
	end
	if IsCrouching or IsSpeeding then
		AccelCur = nil
		return
	end
	
	local hum = GetHumanoid()
	if not hum or hum.Health <= 0 then
		AccelCur = nil
		return
	end
	if hum.MoveDirection.Magnitude < 0.1 then
		AccelCur = nil
		return
	end
	
	local rate = CONFIG.ACCEL_RATE or 32
	local minSp = CONFIG.ACCEL_MIN or 4
	
	if AccelCur == nil then
		local cur = hum.WalkSpeed
		if cur and cur > 1 and cur < 100 then
			NormalWalkSpeed = cur
		end
		AccelCur = math.min(minSp, NormalWalkSpeed)
		SafePcall(function() hum.WalkSpeed = AccelCur end)
	else
		local cur = hum.WalkSpeed
		if cur and math.abs(cur - AccelCur) > 2 and math.abs(cur - NormalWalkSpeed) < 30 then
			if math.abs(cur - AccelCur) > (rate * dt + 1) then
				NormalWalkSpeed = cur
			end
		end
		
		if AccelCur < NormalWalkSpeed then
			AccelCur = math.min(NormalWalkSpeed, AccelCur + rate * dt)
			SafePcall(function() hum.WalkSpeed = AccelCur end)
		else
			if math.abs((cur or 0) - NormalWalkSpeed) > 0.5 then
				SafePcall(function() hum.WalkSpeed = NormalWalkSpeed end)
			end
			AccelCur = NormalWalkSpeed
		end
	end
end

-- Crouch System
local function CooldownLeft()
	local elapsed = os.clock() - LastUncrouchAt
	return math.max(0, (CONFIG.CROUCH_COOLDOWN or 5) - elapsed)
end

local function NormalizeId(s)
	s = tostring(s or "")
	if s == "" then return "" end
	if tonumber(s) then return "rbxassetid://" .. s end
	return s
end

local function StopCrouchTracks()
	SafePcall(function() if CrouchIdleTrack then pcall(function() CrouchIdleTrack:Stop(0.15) end) end end)
	SafePcall(function() if CrouchWalkTrack then pcall(function() CrouchWalkTrack:Stop(0.15) end) end end)
	-- wait out fade so tracks don't linger, then release
	local o1, o2 = CrouchIdleObj, CrouchWalkObj
	CrouchIdleTrack, CrouchWalkTrack = nil, nil
	CrouchIdleObj, CrouchWalkObj = nil, nil
	SafePcall(function() if o1 then o1:Destroy() end end)
	SafePcall(function() if o2 then o2:Destroy() end end)
	
	SafePcall(function()
		local hum = GetHumanoid()
		if hum then
			local an = hum:FindFirstChildOfClass("Animator")
			if an then
				for _, tr in ipairs(an:GetPlayingAnimationTracks()) do
					if tr.Name == "CrouchIdle" or tr.Name == "CrouchWalk" then
						tr:Stop(0.15)
					end
				end
			end
		end
	end)
end

local function ExitCrouch(forced, keepSpeed)
	if not IsCrouching then return end
	IsCrouching = false
	StopCrouchTracks()
	
	SafePcall(function()
		local hum = GetHumanoid()
		if hum then
			hum.WalkSpeed = OrigWalkSpeed
			NormalWalkSpeed = OrigWalkSpeed
			AccelCur = nil
		end
	end)
	
	local _returnToSpeed = WasSpeedBeforeCrouch and CONFIG.SPEED_ENABLED
	WasSpeedBeforeCrouch = false
	LastUncrouchAt = os.clock()
	
	if _returnToSpeed then
		SafePcall(function()
			local hum2 = GetHumanoid()
			local target = CONFIG.SPEED_VALUE or 28
			if hum2 then
				hum2.WalkSpeed = target
				NormalWalkSpeed = target
				OrigWalkSpeed = target
				IsSpeeding = true
				AccelCur = nil
				if SpeedBtn then
					SpeedBtn.BackgroundColor3 = Color3.fromRGB(50, 130, 200)
					SpeedBtn.Text = "SPEED: ON"
				end
			end
		end)
	end
	
	SafePcall(function()
		if CrouchBtn then
			CrouchBtn.BackgroundColor3 = Color3.fromRGB(40, 40, 50)
			CrouchBtn.Text = "CROUCH"
		end
	end)
end

local function EnterCrouch()
	if IsCrouching then
		ExitCrouch(false)
		return
	end
	if not CONFIG.CROUCH_ENABLED then return end
	
	WasSpeedBeforeCrouch = IsSpeeding
	if IsSpeeding then ExitSpeed() end
	if CooldownLeft() > 0.05 then return end
	
	local hum = GetHumanoid()
	if not hum or hum.Health <= 0 then return end
	
	OrigWalkSpeed = hum.WalkSpeed
	NormalWalkSpeed = hum.WalkSpeed
	if OrigWalkSpeed > (CONFIG.CROUCH_BREAK_SPEED or 7) + 5 or OrigWalkSpeed < 1 then
		OrigWalkSpeed = 16
		NormalWalkSpeed = 16
	end
	
	StopCrouchTracks()
	IsCrouching = true
	SafePcall(function() if DespMode ~= "None" then StopDesperationTracks() end end) -- Exclusive: cut Desp instantly so crouch shows
	SafePcall(function() hum.WalkSpeed = CONFIG.CROUCH_SPEED or 5 end)
	
	SafePcall(function()
		local an = hum:FindFirstChildOfClass("Animator") or hum:WaitForChild("Animator", 2)
		local idleId = NormalizeId(CONFIG.CROUCH_ANIM_ID)
		local walkId = NormalizeId(CONFIG.CROUCH_WALK_ID)
		
		if idleId ~= "" then
			local a1 = Instance.new("Animation")
			a1.AnimationId = idleId
			local t1 = an:LoadAnimation(a1)
			t1.Name = "CrouchIdle"
			t1.Priority = Enum.AnimationPriority.Action
			t1.Looped = true
			CrouchIdleObj, CrouchIdleTrack = a1, t1
			t1:Play(0.15)
		end
		
		if walkId ~= "" then
			local a2 = Instance.new("Animation")
			a2.AnimationId = walkId
			local t2 = an:LoadAnimation(a2)
			t2.Name = "CrouchWalk"
			t2.Priority = Enum.AnimationPriority.Action
			t2.Looped = true
			CrouchWalkObj, CrouchWalkTrack = a2, t2
			t2:Play(0.15)
		end
	end)
	
	SafePcall(function()
		if CrouchBtn then
			CrouchBtn.BackgroundColor3 = Color3.fromRGB(50, 150, 80)
			CrouchBtn.Text = "UNCROUCH"
		end
	end)
end

local function UpdateCrouch(dt)
	local hum = GetHumanoid()
	if not hum then
		if IsCrouching then
			IsCrouching = false
			StopCrouchTracks()
		end
		return
	end
	
	if not CONFIG.CROUCH_ENABLED and IsCrouching then
		ExitCrouch(true)
		return
	end
	
	if IsSpeeding and IsCrouching then
		ExitCrouch(true)
		return
	end
	
	if IsCrouching then
		local brk = CONFIG.CROUCH_BREAK_SPEED or 7
		local vel = 0
		SafePcall(function()
			vel = hum.RootPart and hum.RootPart.AssemblyLinearVelocity.Magnitude or 0
		end)
		
		if hum.WalkSpeed > brk + 0.5 or vel > brk + 1.5 then
			ExitCrouch(true)
			return
		end
		
		if math.abs(hum.WalkSpeed - (CONFIG.CROUCH_SPEED or 5)) > 0.1 then
			SafePcall(function() hum.WalkSpeed = CONFIG.CROUCH_SPEED or 5 end)
		end
		
		SafePcall(function()
			-- Re-play if tracks died while Desp was stopping/starting (fixes invisible crouch in Last/Final)
			local cmult = tonumber(CONFIG.CROUCH_ANIM_SPEED) or 1
			if CrouchIdleTrack and not CrouchIdleTrack.IsPlaying then pcall(function() CrouchIdleTrack:Play(0.15, 1, cmult) end) end
			if CrouchWalkTrack and not CrouchWalkTrack.IsPlaying then pcall(function() CrouchWalkTrack:Play(0.15, 1, cmult) end) end
			local moving = hum.MoveDirection.Magnitude > 0.1
			if CrouchIdleTrack then
				if moving then
					CrouchIdleTrack:AdjustWeight(0.05)
				else
					CrouchIdleTrack:AdjustWeight(1)
					if math.abs(CrouchIdleTrack.Speed - cmult) > 0.1 then
						CrouchIdleTrack:AdjustSpeed(cmult)
					end
				end
			end
			
			if CrouchWalkTrack then
				if moving then
					CrouchWalkTrack:AdjustWeight(1)
					local backward = IsMovingBackward()
					local mult = tonumber(CONFIG.BACKWARD_ANIM_SPEED) or 1
					local wantSpd = cmult
					if backward then wantSpd = -1 * mult * cmult end
					if math.abs(CrouchWalkTrack.Speed - wantSpd) > 0.1 then CrouchWalkTrack:AdjustSpeed(wantSpd) end
				else
					CrouchWalkTrack:AdjustWeight(0.05)
				end
			end
		end)
	end
end


-- Desperation Anims (Last / Final Chance, backwards-compatible)
StopDesperationTracks = function()
	SafePcall(function() if DespIdleTrack then pcall(function() DespIdleTrack:Stop(0.15) end) end end)
	SafePcall(function() if DespWalkTrack then pcall(function() DespWalkTrack:Stop(0.15) end) end end)
	SafePcall(function() if DespRunTrack then pcall(function() DespRunTrack:Stop(0.15) end) end end)
	local o1, o2, o3 = DespIdleObj, DespWalkObj, DespRunObj
	DespIdleTrack, DespWalkTrack, DespRunTrack = nil, nil, nil
	DespIdleObj, DespWalkObj, DespRunObj = nil, nil, nil
	DespHum = nil
	DespMode = "None"
	DespActiveState = "None"
	_G.__DesperationActive = false
	SafePcall(function() if o1 then o1:Destroy() end end)
	SafePcall(function() if o2 then o2:Destroy() end end)
	SafePcall(function() if o3 then o3:Destroy() end end)
	SafePcall(function()
		local hum = GetHumanoid()
		if hum then
			local an = hum:FindFirstChildOfClass("Animator")
			if an then
				for _, tr in ipairs(an:GetPlayingAnimationTracks()) do
					if tr.Name == "DespIdle" or tr.Name == "DespWalk" or tr.Name == "DespRun" then
						tr:Stop(0.15)
					end
				end
			end
		end
	end)
end

local function EnsureDesperationTracks(hum)
	if not hum then return false end
	if DespHum == hum and DespIdleTrack and DespWalkTrack and DespRunTrack then return true end
	StopDesperationTracks()
	local ok = false
	SafePcall(function()
		local an = hum:FindFirstChildOfClass("Animator") or hum:WaitForChild("Animator", 2)
		if not an then return end
		local function mk(idStr, nm, prio)
			local id = NormalizeId(idStr)
			if id == "" then return nil, nil end
			local a = Instance.new("Animation")
			a.AnimationId = id
			local t = an:LoadAnimation(a)
			t.Name = nm
			t.Priority = prio
			t.Looped = true
			return a, t
		end
		local a1, t1 = mk(CONFIG.DESP_IDLE_ID, "DespIdle", Enum.AnimationPriority.Action)
		local a2, t2 = mk(CONFIG.DESP_WALK_ID, "DespWalk", Enum.AnimationPriority.Action)
		local a3, t3 = mk(CONFIG.DESP_RUN_ID, "DespRun", Enum.AnimationPriority.Action)
		if not (t1 and t2 and t3) then
			if a1 then pcall(function() a1:Destroy() end) end
			if a2 then pcall(function() a2:Destroy() end) end
			if a3 then pcall(function() a3:Destroy() end) end
			return
		end
		DespIdleObj, DespIdleTrack = a1, t1
		DespWalkObj, DespWalkTrack = a2, t2
		DespRunObj, DespRunTrack = a3, t3
		DespHum = hum
		DespMode = "None"
		ok = true
	end)
	return ok
end

local function GetDespHPState()
	if CONFIG.DESP_USE_HP == false then return nil end
	if _G.DesperationForce == "Last" then return "Last" end
	if _G.DesperationForce == "Final" then return "Final" end
	local hum = GetHumanoid()
	if not hum or (hum.MaxHealth or 0) <= 0 then return nil end
	local pct = (hum.Health / hum.MaxHealth) * 100
	local finalAt = tonumber(CONFIG.DESP_FINAL_HP) or 20
	local lastAt = tonumber(CONFIG.DESP_LAST_HP) or 40
	if finalAt > lastAt then finalAt = lastAt end
	if pct <= finalAt then return "Final"
	elseif pct <= lastAt then return "Last" end
	return nil
end

local function UpdateDesperationAnims(stateKey)
	local forced = _G.DesperationForce
	local want = nil
	if forced == "Final" then want = "Final"
	elseif forced == "Last" then want = "Last"
	elseif forced == "None" then
		if DespMode ~= "None" then StopDesperationTracks() end
		return
	elseif stateKey == "final" then want = "Final"
	elseif stateKey == "lastchance" or stateKey == "lastlayer" then want = "Last"
	end
	if not want then want = GetDespHPState() end
	if not want then
		if DespMode ~= "None" then StopDesperationTracks() end
		DespActiveState = "None"
		return
	end
	if not CONFIG.DESP_ENABLED then
		if DespMode ~= "None" then StopDesperationTracks() end
		return
	end
	if IsDead then
		if DespMode ~= "None" then StopDesperationTracks() end
		return
	end
	local crouched = (IsCrouching == true)
	if crouched then
		-- Exclusive: crouch wins, kill Desp entirely so crouch anim is visible (no blend)
		if DespMode ~= "None" then StopDesperationTracks() end
		DespActiveState = "None"
		_G.__DesperationActive = false
		return
	end
	local hum = GetHumanoid()
	if not hum or hum.Health <= 0 then
		if DespMode ~= "None" then StopDesperationTracks() end
		return
	end
	if not EnsureDesperationTracks(hum) then return end
	local hrp = GetRootPart()
	local speed, moving = 0, false
	SafePcall(function()
		local md = hum.MoveDirection
		if md.Magnitude > 0.1 then
			moving = true
			if hrp then
				local v = hrp.AssemblyLinearVelocity * Vector3.new(1, 0, 1)
				speed = v.Magnitude
			end
			if speed < 0.5 then speed = md.Magnitude * (hum.WalkSpeed or 16) end
		end
	end)
	local mode = "Idle"
	local runAt = tonumber(CONFIG.DESP_RUN_MIN_SPEED) or 18
	local runHold = runAt - 2
	if DespMode == "Run" and moving and speed >= runHold then
		mode = "Run"
	elseif moving and speed >= runAt then
		mode = "Run"
	elseif moving then
		mode = "Walk"
	end
	local backward = false
	SafePcall(function() backward = IsMovingBackward() end)
	local baseMult = 1
	if mode == "Idle" then baseMult = tonumber(CONFIG.DESP_IDLE_SPEED) or 1
	elseif mode == "Walk" then baseMult = tonumber(CONFIG.DESP_WALK_SPEED) or 1
	elseif mode == "Run" then baseMult = tonumber(CONFIG.DESP_RUN_SPEED) or 1 end
	local bmult = tonumber(CONFIG.BACKWARD_ANIM_SPEED) or 1
	local spd = 1 * baseMult
	if backward and moving then spd = -1 * bmult * baseMult end
	DespActiveState = want
	_G.__DesperationActive = true
	-- Exclusive (no blend): Desp always full weight Action when standing. Crouched path returned early above.
	SafePcall(function()
		if DespIdleTrack then DespIdleTrack.Priority = Enum.AnimationPriority.Action end
		if DespWalkTrack then DespWalkTrack.Priority = Enum.AnimationPriority.Action end
		if DespRunTrack then DespRunTrack.Priority = Enum.AnimationPriority.Action end
	end)
	local topW = 1
	SafePcall(function()
		local map = { Idle = DespIdleTrack, Walk = DespWalkTrack, Run = DespRunTrack }
		for name, tr in pairs(map) do
			if tr then
				if name == mode then
					if not tr.IsPlaying then tr:Play(0.15, 1, spd) end
					tr:AdjustWeight(topW)
					if math.abs(tr.Speed - spd) > 0.05 then tr:AdjustSpeed(spd) end
				else
					if tr.IsPlaying then tr:AdjustWeight(0.05) end
				end
			end
		end
	end)
	DespMode = mode
end

-- Speed System
function ExitSpeed(silent, isAutoBreak)
	if not IsSpeeding then
		if not isAutoBreak then
			SpeedAutoBroken = false
			SpeedBreakBelowTime = 0
		end
		return
	end
	
	IsSpeeding = false
	if not isAutoBreak then
		SpeedAutoBroken = false
		SpeedBreakBelowTime = 0
	end
	AccelCur = nil
	
	SafePcall(function()
		local hum = GetHumanoid()
		if hum then
			local back = OrigWalkSpeed
			if not back or back < 1 or back > 60 then back = 16 end
			hum.WalkSpeed = back
			NormalWalkSpeed = back
		end
	end)
	
	SafePcall(function()
		if SpeedBtn then
			SpeedBtn.BackgroundColor3 = Color3.fromRGB(40, 40, 50)
			SpeedBtn.Text = "SPEED: OFF"
		end
	end)
end

function EnterSpeed()
	if IsSpeeding then
		ExitSpeed()
		return
	end
	
	if SpeedAutoBroken then
		SpeedAutoBroken = false
		SpeedBreakBelowTime = 0
	end
	if not CONFIG.SPEED_ENABLED then return end
	
	local hum = GetHumanoid()
	if not hum or hum.Health <= 0 then return end
	
	if IsCrouching then ExitCrouch(true) end
	
	local cur = hum.WalkSpeed
	if cur and cur >= 1 and cur <= 60 and cur < (CONFIG.SPEED_VALUE or 28) then
		OrigWalkSpeed = cur
	elseif not OrigWalkSpeed or OrigWalkSpeed < 1 or OrigWalkSpeed > 60 then
		OrigWalkSpeed = 16
	end
	
	local target = CONFIG.SPEED_VALUE or 28
	NormalWalkSpeed = target
	AccelCur = nil
	IsSpeeding = true
	
	SafePcall(function() hum.WalkSpeed = target end)
	SafePcall(function()
		if SpeedBtn then
			SpeedBtn.BackgroundColor3 = Color3.fromRGB(50, 130, 200)
			SpeedBtn.Text = "SPEED: ON"
		end
	end)
end

function UpdateSpeed(dt)
	local hum = GetHumanoid()
	
	if SpeedAutoBroken then
		if not CONFIG.SPEED_BREAK_ENABLED then
			SpeedAutoBroken = false
			SpeedBreakBelowTime = 0
			return
		end
		if not CONFIG.SPEED_BREAK_AUTO_RESUME then return end
		if not CONFIG.SPEED_ENABLED then
			SpeedAutoBroken = false
			SpeedBreakBelowTime = 0
			return
		end
		if IsCrouching then
			SpeedBreakBelowTime = 0
			return
		end
		if not hum or hum.Health <= 0 then
			SpeedBreakBelowTime = 0
			return
		end
		
		local limit = CONFIG.SPEED_BREAK_LIMIT or 30
		local hyst = CONFIG.SPEED_BREAK_HYSTERESIS or 2
		local resumeUnder = limit - hyst
		local vel = 0
		
		SafePcall(function()
			vel = hum.RootPart and hum.RootPart.AssemblyLinearVelocity.Magnitude or 0
		end)
		
		local cur = hum.WalkSpeed or 0
		if cur <= resumeUnder and vel <= resumeUnder + 1.5 then
			SpeedBreakBelowTime = SpeedBreakBelowTime + (dt or 0.016)
			if SpeedBreakBelowTime >= (CONFIG.SPEED_BREAK_RESUME_DELAY or 0.5) then
				SpeedAutoBroken = false
				SpeedBreakBelowTime = 0
				SafePcall(function() EnterSpeed() end)
				if not IsSpeeding and CONFIG.SPEED_ENABLED then
					SpeedAutoBroken = true
					SpeedBreakBelowTime = 0
				end
			end
		else
			SpeedBreakBelowTime = 0
		end
		return
	end
	
	if not IsSpeeding then return end
	if not CONFIG.SPEED_ENABLED then
		ExitSpeed()
		return
	end
	if not hum or hum.Health <= 0 then
		IsSpeeding = false
		AccelCur = nil
		return
	end
	if IsCrouching then
		ExitSpeed()
		return
	end
	
	local target = CONFIG.SPEED_VALUE or 28
	NormalWalkSpeed = target
	
	if CONFIG.SPEED_BREAK_ENABLED then
		local limit = CONFIG.SPEED_BREAK_LIMIT or 30
		local vel = 0
		SafePcall(function()
			vel = hum.RootPart and hum.RootPart.AssemblyLinearVelocity.Magnitude or 0
		end)
		
		local cur = hum.WalkSpeed or target
		local pushedOver = (cur > limit + 0.5 and (cur - target) > 1.5)
		local velOver = (vel > limit + 1.5)
		
		if pushedOver or velOver then
			SafePcall(function() ExitSpeed() end)
			SpeedAutoBroken = true
			SpeedBreakBelowTime = 0
			warn("[Chase] SPEED BREAK: spike over LIMIT " .. tostring(limit))
			return
		end
	end
	
	if math.abs(hum.WalkSpeed - target) > 0.5 then
		SafePcall(function() hum.WalkSpeed = target end)
	end
end

-- Ray Radar System (merged, PNG picture-only like Crouch)
local function RadarReady() return os.clock() >= (RadarCooldownUntil or 0) end
local function RadarOn() return os.clock() < (RadarActiveUntil or 0) end
local function RadarCooldownLeft()
  return math.max(0, (RadarCooldownUntil or 0) - os.clock())
end
local function TriggerRadar()
  if not CONFIG.RADAR_ENABLED then return end
  if not RadarReady() then return end
  RadarActiveUntil = os.clock() + (tonumber(CONFIG.RADAR_DURATION) or 5)
  RadarCooldownUntil = os.clock() + (tonumber(CONFIG.RADAR_COOLDOWN) or 10)
end
local RadarGuiHolder = nil
local function GetRadarHolder()
  if RadarGuiHolder and RadarGuiHolder.Parent then return RadarGuiHolder end
  local pg = nil
  pcall(function() pg = LocalPlayer:WaitForChild("PlayerGui", 10) end)
  if not pg then return nil end
  local f = pg:FindFirstChild("RadarHLHolder")
  if not f then
    f = Instance.new("Folder")
    f.Name = "RadarHLHolder"
    f.Parent = pg
  end
  RadarGuiHolder = f
  return f
end
local function GetRadarHL(plr)
  local h = RadarHighlights[plr]
  -- revalidate: destroyed, deparented, or adornee pointing at dead model
  if h then
    local ok = false
    pcall(function() ok = (h.Parent ~= nil) end)
    if not ok then h = nil RadarHighlights[plr] = nil end
  end
  if h then return h end
  local holder = GetRadarHolder()
  if not holder then return nil end
  local nh = Instance.new("Highlight")
  nh.Name = "RadarHL_" .. tostring(plr.UserId)
  nh.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
  nh.FillTransparency = 1
  nh.OutlineTransparency = 1
  nh.Enabled = true
  pcall(function()
    nh.Adornee = plr.Character
    nh.Parent = holder
  end)
  RadarHighlights[plr] = nh
  -- if their character respawns later, re-point adornee
  local hooked = false pcall(function() hooked = plr:GetAttribute("RadarHooked") end) if not hooked then
    pcall(function() plr:SetAttribute("RadarHooked", true) end)
    TrackConn(plr.CharacterAdded:Connect(function(c)
      task.delay(0.5, function()
        SafePcall(function()
          local hh = RadarHighlights[plr]
          if hh and hh.Parent then
            hh.Adornee = c
          else
            RadarHighlights[plr] = nil
            if CONFIG.RADAR_AUTO_REFRESH ~= false then
              local h2 = GetRadarHL(plr)
              if h2 and c then pcall(function() h2.Adornee = c end) end
            end
          end
        end)
      end)
    end))
  end
  return nh
end
-- Full rebuild: recreates any missing/destroyed highlights (late joiners, respawns, streamed chars)
local function RefreshRadarESP()
  SafePcall(function()
    local holder = GetRadarHolder()
    if not holder then return end
    for _, plr in ipairs(Players:GetPlayers()) do
      if plr ~= LocalPlayer then
        local h = RadarHighlights[plr]
        local dead = true
        pcall(function() dead = (h == nil or h.Parent == nil) end)
        if dead then
          RadarHighlights[plr] = nil
          GetRadarHL(plr)
        else
          -- re-point adornee in case character respawned (old adornee destroyed)
          local ch = plr.Character
          if ch then
            local cur = nil
            pcall(function() cur = h.Adornee end)
            if cur ~= ch then
              local hrp = ch:FindFirstChild("HumanoidRootPart")
              if hrp then pcall(function() h.Adornee = ch end) end
            end
          end
        end
      end
    end
  end)
end
TrackConn(Players.PlayerRemoving:Connect(function(p)
  local h = RadarHighlights[p]
  if h then pcall(function() h:Destroy() end) end
  RadarHighlights[p] = nil
end))
TrackConn(Players.PlayerAdded:Connect(function(p)
  task.delay(1, function()
    SafePcall(function()
      if CONFIG.RADAR_AUTO_REFRESH ~= false then RefreshRadarESP() end
    end)
  end)
  pcall(function()
    local hooked2 = false pcall(function() hooked2 = p:GetAttribute("RadarHooked") end) if not hooked2 then
      p:SetAttribute("RadarHooked", true)
      TrackConn(p.CharacterAdded:Connect(function(c)
        task.delay(0.5, function()
          SafePcall(function()
            local hh = RadarHighlights[p]
            if hh and hh.Parent then hh.Adornee = c else RadarHighlights[p] = nil local h2 = GetRadarHL(p) if h2 and c then pcall(function() h2.Adornee = c end) end end
          end)
        end)
      end))
    end
  end)
end))
local function UpdateRadar()
  local mr = GetRootPart()
  local d = CONFIG.LAYER_DISTANCES or {116,75,57}
  local far, mid, near = tonumber(d[1]) or 116, tonumber(d[2]) or 75, tonumber(d[3]) or 57
  local on = RadarOn()
  for _, plr in ipairs(Players:GetPlayers()) do
    if plr ~= LocalPlayer then
    local char = plr.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not hrp or not hum or hum.Health <= 0 or not on then
      local h0 = RadarHighlights[plr]
      if h0 then pcall(function() h0.FillTransparency = 1 h0.OutlineTransparency = 1 end) end
    else
    local dist = mr and (mr.Position - hrp.Position).Magnitude or 9999
    local col, inRange
    if dist <= near then col, inRange = Color3.fromRGB(255,0,0), true
    elseif dist <= mid then col, inRange = Color3.fromRGB(255,255,0), true
    elseif dist <= far then col, inRange = Color3.fromRGB(0,255,0), true
    else col, inRange = Color3.fromRGB(0,255,0), false end
    local h = GetRadarHL(plr)
    if h then
    pcall(function()
      if h.Adornee ~= char then h.Adornee = char end
      h.FillColor = col
      h.OutlineColor = col
      if inRange then
        h.FillTransparency = tonumber(CONFIG.RADAR_IN_TRANS) or 0.15
        h.OutlineTransparency = 0
      else
        local tt = tonumber(CONFIG.RADAR_OUT_TRANS) or 0.9
        h.FillTransparency = tt
        h.OutlineTransparency = tt
      end
    end)
    end
    end
    end
  end
end
local function RefreshRadarBtn()
  if not RadarBtn then return end
  SafePcall(function()
    local cd = RadarCooldownLeft()
    if cd > 0.05 then
      local total = tonumber(CONFIG.RADAR_COOLDOWN) or 10
      local frac = Clamp(cd / math.max(0.01,total), 0, 1)
      local idx = Clamp(math.ceil(frac * 10), 1, 10)
      for i, im in ipairs(RadarCdImgs) do
        local show = (i == idx)
        im.Visible = show
        if show then pcall(function() im.ImageTransparency = (CONFIG.CD_IMG_TRANSPARENCY or 0.5) end) end
      end
      if RadarBtnImg then RadarBtnImg.Visible = false end
      RadarBtn.Text = ""
      RadarBtn.BackgroundColor3 = Color3.fromRGB(90,90,90)
      RadarBtn.BackgroundTransparency = 1
      -- 50% transparent on cooldown
      SafePcall(function()
        for _, im in ipairs(RadarCdImgs) do
          if im.Visible then im.ImageTransparency = (CONFIG.CD_IMG_TRANSPARENCY or 0.5) end
        end
      end)
    else
      for _, im in ipairs(RadarCdImgs) do
        im.Visible = false
        pcall(function() im.ImageTransparency = 0 end)
      end
      if RadarBtnImg and RadarBtnImg.Image ~= "" then
        RadarBtnImg.Visible = true
        pcall(function() RadarBtnImg.ImageTransparency = 0 end)
        RadarBtn.Text = ""
      else
        if RadarBtnImg then RadarBtnImg.Visible = false end
        if RadarOn() then RadarBtn.Text = "ON" else RadarBtn.Text = "RAY" end
      end
      if RadarOn() then RadarBtn.BackgroundColor3 = Color3.fromRGB(50,150,80)
      else RadarBtn.BackgroundColor3 = Color3.fromRGB(40,40,50) end
      -- picture-only: no black bg when image present
      if RadarBtnImg and RadarBtnImg.Visible then RadarBtn.BackgroundTransparency = 1
      else RadarBtn.BackgroundTransparency = 0.15 end
    end
    RadarBtn.Visible = (CONFIG.RADAR_SHOW_BUTTON ~= false) and (CONFIG.RADAR_ENABLED ~= false)
  end)
end

-- SetSpawn Ability System (GitHub PNG picture-only like Crouch/Ray)
local function SetCharTransparent(char)
  if not char then return end
  pcall(function()
    for _, d in ipairs(char:GetDescendants()) do
      if d:IsA("BasePart") then d.Transparency = 1
      elseif d:IsA("Decal") or d:IsA("Texture") then d.Transparency = 1
      elseif d:IsA("BillboardGui") or d:IsA("SurfaceGui") then pcall(function() d.Enabled = false end) end
    end
  end)
end
local function HookSpawnTransparency(char)
  if SpawnTransparentHook then pcall(function() SpawnTransparentHook:Disconnect() end) SpawnTransparentHook = nil end
  if not char then return end
  SpawnTransparentHook = char.DescendantAdded:Connect(function(d)
    if not SpawnTransparentActive then return end
    task.delay(0.05, function()
      pcall(function()
        if d:IsA("BasePart") then d.Transparency = 1
        elseif d:IsA("Decal") or d:IsA("Texture") then d.Transparency = 1 end
      end)
    end)
  end)
  TrackConn(SpawnTransparentHook)
end
local function PlaySpawnAnim(char, idOverride)
  if not char then return nil end
  local track = nil
  pcall(function()
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum then return end
    local anim = Instance.new("Animation")
    anim.AnimationId = idOverride or CONFIG.SPAWN_ANIM_ID or "rbxassetid://112757618232626"
    local animator = hum:FindFirstChildOfClass("Animator")
    if animator then track = animator:LoadAnimation(anim) else track = hum:LoadAnimation(anim) end
    if track then track.Priority = Enum.AnimationPriority.Action4 track:Play(0.1, 1, 1) end
  end)
  return track
end
local function TriggerSpawnSet()
  if not CONFIG.SPAWN_ENABLED then return end
  if SpawnUsed then return end
  local char = LocalPlayer.Character
  local hrp = char and char:FindFirstChild("HumanoidRootPart") or nil
  if not hrp or not char then return end
  local hum = char:FindFirstChildOfClass("Humanoid")
  if not hum or hum.Health <= 0 then return end
  SavedSpawnCF = hrp.CFrame
  SpawnArmed = true
  SpawnUsed = true
  RefreshSpawnBtn()
  -- New Set Spawn animation: 50% chance to use the alternate animation
  -- (SPAWN_ANIM2_ID). The original SPAWN_ANIM_ID still plays the other 50%,
  -- so existing spawn behaviour is preserved.
  local spawnAnimId = CONFIG.SPAWN_ANIM_ID
  do
    local altId = CONFIG.SPAWN_ANIM2_ID
    local altChance = tonumber(CONFIG.SPAWN_ANIM2_CHANCE) or 0.5
    if altChance < 0 then altChance = 0 elseif altChance > 1 then altChance = 1 end
    if altId and altId ~= "" and math.random() < altChance then
      spawnAnimId = altId
    end
  end
  local spTrack = PlaySpawnAnim(char, spawnAnimId)
  -- ghost (invisible) effect is optional; off by default so you stay visible
  if CONFIG.SPAWN_GHOST then
    SpawnTransparentActive = true
    -- let the set-spawn animation play visibly before the ghost (transparent) effect kicks in
    local hideDelay = 1.5
    if spTrack and spTrack.Length and spTrack.Length > 0 then hideDelay = math.min(spTrack.Length, 4) end
    task.delay(hideDelay, function()
      if SpawnTransparentActive and LocalPlayer.Character == char then
        SetCharTransparent(char)
        HookSpawnTransparency(char)
      end
    end)
  end
  print("[SetSpawn] armed at " .. tostring(SavedSpawnCF))
end

  SpawnArmed = false
  SpawnUsed = false
  SpawnTransparentActive = false
  if SpawnTransparentHook then pcall(function() SpawnTransparentHook:Disconnect() end) SpawnTransparentHook = nil end
  SafePcall(function() RefreshSpawnBtn() end)
end

local AdrenCdImgs = {}

local FlashTransition = nil
local function IsAdrenActive() return os.clock() < (AdrenActiveUntil or 0) end
local function AdrenReady() return os.clock() >= (AdrenCooldownUntil or 0) end
local function TriggerAdrenaline()
  if not CONFIG.ADREN_ENABLED then return end
  if not AdrenReady() then return end
  if IsAdrenActive() then return end
  local char = LocalPlayer.Character
  local hum = char and char:FindFirstChildOfClass("Humanoid") or nil
  if not hum or hum.Health <= 0 then return end
  if IsDead then return end
  local dur = tonumber(CONFIG.ADREN_DURATION) or 5
  local cd = tonumber(CONFIG.ADREN_COOLDOWN) or 20
  if dur < 1 then dur = 1 end
  if cd < dur then cd = dur end
  -- remember pre-rush speed so it can be restored when the rush ends
  AdrenPrevSpeed = hum.WalkSpeed
  if IsCrouching then
    local nw = tonumber(NormalWalkSpeed)
    if nw and nw >= 1 and nw <= 60 then AdrenPrevSpeed = nw end
  end
  AdrenActiveUntil = os.clock() + dur
  -- cooldown begins when the rush ends so the 20-frame CD plays fully (20 -> 1)
  AdrenCooldownUntil = os.clock() + dur + cd
  if IsCrouching then
    pcall(function()
      IsCrouching = false
      StopCrouchTracks()
    end)
  end
  SafePcall(function()
    local aid = tostring(CONFIG.ADREN_ANIM_ID or "")
    if aid ~= "" then
      local anim = Instance.new("Animation")
      anim.AnimationId = aid
      local animator = hum:FindFirstChildOfClass("Animator")
      local tr = nil
      if animator then tr = animator:LoadAnimation(anim) else tr = hum:LoadAnimation(anim) end
      if tr then tr.Priority = Enum.AnimationPriority.Action4 local asp = _G.ADREN_SPEED_OVERRIDE or tonumber(CONFIG.ADREN_ANIM_SPEED) or 1.8 tr:Play(0.15, 1, asp) AdrenAnimTrack = tr end
    end
  end)
  SafePcall(function() FlashTransition() end)
  SafePcall(function() RefreshAdrenBtn() end)
  print("[Adren] RUSH for " .. tostring(dur) .. "s")
end
local function UpdateAdrenaline(dt)
  local hum = GetHumanoid()
  if IsAdrenActive() and hum and hum.Health > 0 and not IsDead then
    local sp = tonumber(CONFIG.ADREN_SPEED) or 28
    SafePcall(function() hum.WalkSpeed = sp end)
  else
    if AdrenAnimTrack then
      SafePcall(function() AdrenAnimTrack:Stop(0.2) end)
      AdrenAnimTrack = nil
    end
    if AdrenPrevSpeed ~= nil then
      local back = tonumber(AdrenPrevSpeed) or 16
      AdrenPrevSpeed = nil
      if not IsSpeeding then
        if back < 1 or back > 60 then back = 16 end
        SafePcall(function()
          if hum then
            hum.WalkSpeed = back
            NormalWalkSpeed = back
          end
        end)
      end
    end
  end
end

-- ================= v3.0 TP TO SPAWN ABILITY =================
-- "TP to Spawn": plays an ability animation, then teleports the player to a
-- spawn point after a short delay. Two modes: "Closest" (nearest spawn point)
-- and "Random" (randomly selected spawn point), toggled by a separate button.
-- Wrapped in a single init function so it adds only ONE top-level local
-- (keeps the main chunk well under Luau's 200-local limit).
local function InitTpSpawnSystem()
  -- Runtime state mirrors (globals so GUI / keybinds / cleanup can reach them)
  TpSpawnBtn, TpSpawnBtnImg = nil, nil
  TpSpawnCdImgs = {}
  TpSpawnCooldownUntil, TpSpawnPendingUntil = 0, 0
  TpSpawnAnimTrack = nil
  TpModeBtn, TpModeBtnImg, TpModeLbl = nil, nil, nil
  -- Resolved asset URLs for the mode-toggle button (default vs cooldown state)
  local tpModeAsset, tpModeCdAsset = "", ""

  -- Try an asset name, then fall back to the opposite extension case.
  -- (The repo mixes .png and .PNG; e.g. Tp-to-spawn.png vs Near-or-no.PNG.)
  local function TryTpAsset(fname)
    if not fname or fname == "" then return "" end
    local a = SafeAsset(ASSET_FOLDER .. fname)
    if a ~= "" then return a end
    local alt = tostring(fname):gsub("%.png$", ".PNG")
    if alt == fname then alt = tostring(fname):gsub("%.PNG$", ".png") end
    if alt ~= fname then
      a = SafeAsset(ASSET_FOLDER .. alt)
      if a ~= "" then return a end
    end
    return ""
  end

  -- Re-attempt loading the TP button images until they resolve. Fixes the
  -- first-run timing race where EnsureAssets() is still downloading the PNGs
  -- from GitHub when the buttons are first built (SafeAsset returns "" and the
  -- icon never appears -- same class of bug the Adrenaline retry loop fixed).
  -- Bounded so it stops once everything is loaded (or after ~60s of ticks).
  local tpIconRetries = 0
  local function EnsureTpIcons()
    if tpIconRetries >= 120 then return end
    local allLoaded = true
    -- Base ability icon (Tp-to-spawn.png)
    if TpSpawnBtnImg and TpSpawnBtnImg.Image == "" then
      local a = TryTpAsset(tostring(CONFIG.TPSPAWN_IMAGE_FILE or "Tp-to-spawn.png"))
      if a ~= "" then
        TpSpawnBtnImg.Image = a
        TpSpawnBtnImg.Visible = true
        if TpSpawnBtn then TpSpawnBtn.Text = "" end
      else
        allLoaded = false
      end
    end
    -- Cooldown frames (Tp-to-spawn-cd-1 .. cd-10)
    for i, im in ipairs(TpSpawnCdImgs or {}) do
      if im.Image == "" and CONFIG.TPSPAWN_CD_FILES and CONFIG.TPSPAWN_CD_FILES[i] then
        local a = TryTpAsset(CONFIG.TPSPAWN_CD_FILES[i])
        if a ~= "" then im.Image = a else allLoaded = false end
      end
    end
    -- Mode-toggle icons (Near-or-no.PNG / Near-or-no-cd-1.PNG)
    if tpModeAsset == "" then
      local a = TryTpAsset(tostring(CONFIG.TPMODE_IMAGE_FILE or "Near-or-no.PNG"))
      if a ~= "" then tpModeAsset = a else allLoaded = false end
    end
    if tpModeCdAsset == "" then
      local a = TryTpAsset(tostring(CONFIG.TPMODE_CD_FILE or "Near-or-no-cd-1.PNG"))
      if a ~= "" then tpModeCdAsset = a else allLoaded = false end
    end
    if allLoaded then
      tpIconRetries = 120  -- everything resolved; stop retrying
    else
      tpIconRetries = tpIconRetries + 1
    end
  end

  -- Normalise the configured mode to a known value
  local function NormalizeMode(m)
    if m == "Random" then return "Random" end
    return "Closest"
  end
  TpSpawnMode = NormalizeMode(CONFIG.TPSPAWN_MODE)
  CONFIG.TPSPAWN_MODE = TpSpawnMode

  -- Gather candidate spawn points: every SpawnLocation in the map, plus the
  -- player's saved Set-Spawn point (if enabled). Cached briefly to avoid
  -- re-scanning the whole DataModel on every use.
  local spawnCache, spawnCacheAt = nil, 0
  local function GetSpawnPoints()
    local now = os.clock()
    if spawnCache and (now - spawnCacheAt) < 5 then return spawnCache end
    local pts = {}
    SafePcall(function()
      for _, d in ipairs(workspace:GetDescendants()) do
        if d:IsA("SpawnLocation") then
          table.insert(pts, { cframe = d.CFrame, pos = d.Position, name = d.Name })
        end
      end
    end)
    if CONFIG.TPSPAWN_INCLUDE_SETSPAWN ~= false and SavedSpawnCF then
      table.insert(pts, { cframe = SavedSpawnCF, pos = SavedSpawnCF.Position, name = "SetSpawn" })
    end
    spawnCache, spawnCacheAt = pts, now
    return pts
  end

  -- Pick a target spawn point according to the current mode
  local function PickSpawnPoint(hrp)
    local pts = GetSpawnPoints()
    if #pts == 0 then return nil end
    if TpSpawnMode == "Random" then
      return pts[math.random(1, #pts)]
    end
    -- "Closest": nearest by distance from the player's root part
    local myPos = hrp.Position
    local best, bestD = nil, math.huge
    for _, p in ipairs(pts) do
      local d = (myPos - p.pos).Magnitude
      if d < bestD then best, bestD = p, d end
    end
    return best
  end

  -- Perform the actual teleport (client-side, single call)
  local function TeleportToSpawn()
    local char = LocalPlayer.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return false end
    local target = PickSpawnPoint(hrp)
    if not target then return false end
    local ok = pcall(function()
      hrp.CFrame = target.cframe + Vector3.new(0, 3, 0)
      hrp.AssemblyLinearVelocity = Vector3.zero
      hrp.AssemblyAngularVelocity = Vector3.zero
    end)
    return ok
  end

  -- Public: activate the ability
  function TriggerTpSpawn()
    if not CONFIG.TPSPAWN_ENABLED then return end
    local now = os.clock()
    if now < (TpSpawnCooldownUntil or 0) then return end
    if now < (TpSpawnPendingUntil or 0) then return end
    local char = LocalPlayer.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then return end
    local cd = tonumber(CONFIG.TPSPAWN_COOLDOWN) or 10
    local delay = tonumber(CONFIG.TPSPAWN_DELAY) or 5
    if delay < 0 then delay = 0 end
    if cd < delay then cd = delay end
    TpSpawnCooldownUntil = now + cd
    TpSpawnPendingUntil = now + delay
    -- Ability animation (plays immediately on activation)
    SafePcall(function()
      local aid = tostring(CONFIG.TPSPAWN_ANIM_ID or "")
      if aid ~= "" then
        local anim = Instance.new("Animation")
        anim.AnimationId = aid
        local animator = hum:FindFirstChildOfClass("Animator")
        local tr = nil
        if animator then tr = animator:LoadAnimation(anim) else tr = hum:LoadAnimation(anim) end
        if tr then
          tr.Priority = Enum.AnimationPriority.Action4
          tr:Play(0.15, 1, 1)
          TpSpawnAnimTrack = tr
        end
      end
    end)
    SafePcall(function() RefreshTpSpawnBtn() end)
    -- Teleport executes after the configured delay
    task.delay(delay, function()
      TpSpawnPendingUntil = 0
      SafePcall(function()
        if TpSpawnAnimTrack then pcall(function() TpSpawnAnimTrack:Stop(0.2) end) TpSpawnAnimTrack = nil end
        TeleportToSpawn()
      end)
      SafePcall(function() RefreshTpSpawnBtn() end)
    end)
    print("[TpSpawn] " .. tostring(TpSpawnMode) .. " -> teleport in " .. tostring(delay) .. "s")
  end

  -- Public: toggle Closest <-> Random
  function ToggleTpSpawnMode()
    TpSpawnMode = (TpSpawnMode == "Random") and "Closest" or "Random"
    CONFIG.TPSPAWN_MODE = TpSpawnMode
    SafePcall(function() SaveConfig(true) end)
    SafePcall(function() RefreshTpModeBtn() end)
    SafePcall(function() RefreshTpSpawnBtn() end)
    print("[TpSpawn] mode = " .. tostring(TpSpawnMode))
  end

  -- Public: read current mode
  function GetTpSpawnMode() return TpSpawnMode end

  -- Refresh the ability button (icon / cooldown frames / visibility)
  function RefreshTpSpawnBtn()
    EnsureTpIcons()   -- lazily (re)load icons in case assets finished downloading
    if not TpSpawnBtn then return end
    SafePcall(function()
      local now = os.clock()
      local cd = math.max(0, (TpSpawnCooldownUntil or 0) - now)
      local pending = now < (TpSpawnPendingUntil or 0)
      for _, im in ipairs(TpSpawnCdImgs) do im.Visible = false pcall(function() im.ImageTransparency = 0 end) end
      if pending then
        -- Charging (waiting for the delayed teleport): dim the base icon
        if TpSpawnBtnImg and TpSpawnBtnImg.Image ~= "" then
          TpSpawnBtnImg.Visible = true
          pcall(function() TpSpawnBtnImg.ImageTransparency = 0.35 end)
          TpSpawnBtn.Text = ""
        end
        TpSpawnBtn.BackgroundTransparency = 1
      elseif cd > 0.05 then
        local total = tonumber(CONFIG.TPSPAWN_COOLDOWN) or 10
        local n = #TpSpawnCdImgs
        if n > 0 then
          local frac = math.clamp(cd / math.max(0.01, total), 0, 1)
          local idx = math.clamp(math.ceil(frac * n), 1, n)
          for i, im in ipairs(TpSpawnCdImgs) do
            im.Visible = (i == idx)
            pcall(function() im.ImageTransparency = (i == idx) and (CONFIG.CD_IMG_TRANSPARENCY or 0.5) or 0 end)
          end
        end
        if TpSpawnBtnImg then TpSpawnBtnImg.Visible = false end
        TpSpawnBtn.Text = ""
        TpSpawnBtn.BackgroundColor3 = Color3.fromRGB(90, 90, 90)
        TpSpawnBtn.BackgroundTransparency = 1
      else
        TpSpawnBtn.BackgroundColor3 = Color3.fromRGB(40, 40, 50)
        if TpSpawnBtnImg and TpSpawnBtnImg.Image ~= "" then
          TpSpawnBtnImg.Visible = true
          pcall(function() TpSpawnBtnImg.ImageTransparency = 0 end)
          TpSpawnBtn.Text = ""
        else
          TpSpawnBtn.Text = "TP"
        end
        TpSpawnBtn.BackgroundTransparency = 1
      end
      TpSpawnBtn.Visible = (CONFIG.TPSPAWN_SHOW_BUTTON ~= false) and (CONFIG.TPSPAWN_ENABLED ~= false)
    end)
    SafePcall(function() RefreshTpModeBtn() end)
  end

  -- Refresh the mode-toggle button (Near-or-no vs Near-or-no-cd-1)
  function RefreshTpModeBtn()
    EnsureTpIcons()   -- lazily (re)load icons in case assets finished downloading
    if not TpModeBtn then return end
    SafePcall(function()
      local onCooldown = os.clock() < (TpSpawnCooldownUntil or 0)
      if TpModeBtnImg then
        local want = onCooldown and tpModeCdAsset or tpModeAsset
        if want and want ~= "" and TpModeBtnImg.Image ~= want then
          TpModeBtnImg.Image = want
        end
        TpModeBtnImg.Visible = (want ~= "")
      end
      if TpModeLbl then
        TpModeLbl.Text = (TpSpawnMode == "Random") and "RAND" or "CLOSE"
      end
      TpModeBtn.Visible = (CONFIG.TPMODE_SHOW_BUTTON ~= false) and (CONFIG.TPSPAWN_ENABLED ~= false)
    end)
  end

  -- Shared button builder for the two TP buttons

local function IsTeamSelected(tn)

GetClosestTarget = function()
	local char = LocalPlayer.Character
	if not char then return nil, math.huge end
	
	local hrp = char:FindFirstChild("HumanoidRootPart")
	if not hrp then return nil, math.huge end
	
	local myPos = hrp.Position
	local best, bestDist = nil, math.huge
	
	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= LocalPlayer then
			local tname = plr.Team and plr.Team.Name or nil
			if IsTeamSelected(tname) then
				local c = plr.Character
				if c and c:FindFirstChild("HumanoidRootPart") then
					local d = (myPos - c.HumanoidRootPart.Position).Magnitude
					if d < bestDist then
						best, bestDist = plr, d
					end
				end
			end
		end
	end
	
	return best, bestDist
end

	local px, py = CONFIG.EYE_POS_X or 0.5, CONFIG.EYE_POS_Y or 0.06
	local vp = Vector2.new(1000, 700)
	local cam = ValidateCamera()
	if cam then
		pcall(function() vp = cam.ViewportSize end)
		if not vp or vp.X < 10 or vp.Y < 10 then vp = Vector2.new(1000, 700) end
	end

		if state == "idle" then
			if "idle" ~= LastLayerKey then
				if LastLayerKey ~= "none" and CONFIG.TRANSITION_ENABLED and TransImg and TransImg.Image ~= "" then
					TransTimer = CONFIG.TRANSITION_DURATION

				LastLayerKey = "idle"

		elseif state == "layer" or state == "lastlayer" then

				if LastLayerKey ~= "none" and CONFIG.TRANSITION_ENABLED and TransImg and TransImg.Image ~= "" then
					TransTimer = CONFIG.TRANSITION_DURATION
				end
				LastLayerKey = key
			end
			if TransTimer > 0 then t = true
			else
				if layerNum <= 1 then a = CONFIG.LAYER1_ENABLED and L1Img and L1Img.Image ~= ""
				elseif layerNum == 2 then b = CONFIG.LAYER2_ENABLED and L2Img and L2Img.Image ~= ""
				else c = CONFIG.LAYER3_ENABLED and L3Img and L3Img.Image ~= "" end
			end
		else
			LastLayerKey = state
		end
	end
	
	if TransTimer > 0 then
		TransTimer = TransTimer - dt
		if TransTimer < 0 then TransTimer = 0 end
	end
	
	SafePcall(function()
		if L1Img then L1Img.Visible = a end
		if L2Img then L2Img.Visible = b end
		if L3Img then L3Img.Visible = c end

-- Audio Functions
ActiveSounds = {}
local function PlaySound(id, vol, looped)
	if not id or id == "" then return nil end
	local parentOk, sndService = pcall(function() return game:GetService("SoundService") end)
	local s = Instance.new("Sound")
	s.SoundId = id
	s.Volume = 0
	s.Looped = looped == true
	s.PlayOnRemove = false
	pcall(function()
		s.Parent = (parentOk and sndService) or workspace
	end)
	table.insert(ActiveSounds, s)
	local ok = SafePcall(function() s:Play() end)
	if not ok then
		pcall(function() s:Destroy() end)
		return nil
	end
	-- failed-load watchdog: destroy silent sounds that never load (prevents ghost leak)
	task.delay(6, function()
		pcall(function()
			if s and s.Parent then
				if not s.IsLoaded and s.TimeLength <= 0 and (s.Volume or 0) <= 0.01 then
					-- still check it's not one of the live slots before killing
					local live = false
					for _, v in pairs(Sounds) do
						if v == s then live = true break end
						if typeof(v) == "table" then
							for _, vv in pairs(v) do if vv == s then live = true break end end
						end
					end
					if not live then s:Destroy() end
				end
			end
		end)
	end)
	SafePcall(function()
		TweenService:Create(s, TweenInfo.new(0.5), {Volume = vol}):Play()
	end)
	return s
end

local function StopSound(obj)
	if not obj then return end
	SafePcall(function()
		TweenService:Create(obj, TweenInfo.new(0.5), {Volume = 0}):Play()
	end)
	task.delay(0.6, function()
		if obj then SafePcall(function() obj:Destroy() end) end
	end)
end

local function UpdateLayers(dist, theme)
	local dists = CONFIG.LAYER_DISTANCES
	local n = 0
	if dist < dists[1] then
		if dist >= dists[2] then n = 1
		elseif dist >= dists[3] then n = 2
		else n = 3 end
	end
	
	for i = 1, 3 do
		local should = i <= n
		local playing = Sounds.layers[i] ~= nil
		if should and not playing then
			local a = {theme.layer1, theme.layer2, theme.layer3}
			Sounds.layers[i] = PlaySound(a[i], CONFIG.VOLUME_LAYERS, true)
		elseif not should and playing then
			StopSound(Sounds.layers[i])
			Sounds.layers[i] = nil
		end
	end
	
	for i = 1, n do
		if Sounds.layers[i] then
			Sounds.layers[i].Volume = CONFIG.VOLUME_LAYERS * (CONFIG.CROSSFADE_FACTOR ^ (n - i))
		end
	end
	
	return n
end

local function UpdateLastLayers(dist, theme)
	if not theme.hasLastLayers then return 0 end
	SafePcall(function() RefreshThemeAssets(theme) end)
	
	local dists = CONFIG.LAST_CHANCE_LAYER_DISTANCES
	local n = 0
	if dist < dists[1] then
		if dist >= dists[2] then n = 1
		elseif dist >= dists[3] then n = 2
		else n = 3 end
	end
	
	for i = 1, 3 do
		local should = i <= n
		local playing = Sounds.lastLayers[i] ~= nil
		if should and not playing then
			local a = {theme.lastLayer1, theme.lastLayer2, theme.lastLayer3}
			Sounds.lastLayers[i] = PlaySound(a[i], CONFIG.VOLUME_LAST_CHANCE, true)
		elseif not should and playing then
			StopSound(Sounds.lastLayers[i])
			Sounds.lastLayers[i] = nil
		end
	end
	
	for i = 1, n do
		if Sounds.lastLayers[i] then
			Sounds.lastLayers[i].Volume = CONFIG.VOLUME_LAST_CHANCE * (CONFIG.CROSSFADE_FACTOR ^ (n - i))
		end
	end
	
	return n
end

local function HookEndedReroll(soundObj, kind)
	if not soundObj or typeof(soundObj) ~= "Instance" then return end
	
	SafePcall(function()
		pcall(function() soundObj.Looped = false end)
		local endedEv = nil
		pcall(function() endedEv = soundObj.Ended end)
		if not endedEv then return end
		
		local ok, _ = pcall(function()
			endedEv:Connect(function()
				local function doReroll()
					if kind == "last" then
						if Sounds.lastChance == soundObj then
							SafePcall(function() soundObj:Destroy() end)
							Sounds.lastChance = nil
							if typeof(math.random) == "function" then
								CurrentLastChanceVariant = math.random(1, 2)
							end
						end
					elseif kind == "final" then
						if Sounds.finalChance == soundObj then
							SafePcall(function() soundObj:Destroy() end)
							Sounds.finalChance = nil
							if typeof(math.random) == "function" then
								CurrentFinalVariant = math.random(1, 2)
							end
						end
					end
				end
				
				if typeof(task) == "table" and typeof(task.delay) == "function" then
					task.delay(0.1, function() SafePcall(doReroll) end)
				elseif typeof(delay) == "function" then
					delay(0.1, function() SafePcall(doReroll) end)
				else
					SafePcall(doReroll)
				end
			end)
		end)
		if not ok then warn("[Chase][Hook] Ended connect failed") end
	end)
end

local function ResolveLastChaseId(theme)
	if not theme then return "" end
	if theme.lastChaseRandom then
		local id = CurrentLastChanceVariant == 1 and theme.lastChase1 or theme.lastChase2
		if id and id ~= "" then return id end
		if theme.lastChase1 and theme.lastChase1 ~= "" then return theme.lastChase1 end
		if theme.lastChase2 and theme.lastChase2 ~= "" then return theme.lastChase2 end
		return ""
	end
	return theme.lastChase or ""
end

	end
	return theme.lastChase and theme.lastChase ~= ""
end

			end
		end
	end
end

local function SetFinalSound(theme)
	if not theme then return end
	SafePcall(function() RefreshThemeAssets(theme) end)
	
	local w = ResolveFinalId(theme)
	if not w or w == "" then return end
	
	if theme and theme.hasFinalCustom then
		if not Sounds.finalChance then
			local s = PlaySound(w, CONFIG.VOLUME_FINAL_CHANCE, true)
			if s then Sounds.finalChance = s end
		else
			local curId = nil
			SafePcall(function() curId = Sounds.finalChance.SoundId end)
			if curId ~= w then
				SafePcall(function() StopSound(Sounds.finalChance) end)
				Sounds.finalChance = nil
				local s2 = PlaySound(w, CONFIG.VOLUME_FINAL_CHANCE, true)
				if s2 then Sounds.finalChance = s2 end
			end
		end
		return
	end
	
	if not Sounds.finalChance then
		local s3 = PlaySound(w, CONFIG.VOLUME_FINAL_CHANCE, false)
		if s3 then
			Sounds.finalChance = s3
			SafePcall(function() HookEndedReroll(s3, "final") end)
		end
	else
		local curId2 = nil
		SafePcall(function() curId2 = Sounds.finalChance.SoundId end)
		if curId2 ~= w then
			SafePcall(function() StopSound(Sounds.finalChance) end)
			Sounds.finalChance = nil
			local s4 = PlaySound(w, CONFIG.VOLUME_FINAL_CHANCE, false)
			if s4 then
				Sounds.finalChance = s4
				SafePcall(function() HookEndedReroll(s4, "final") end)
			end
		end
	end
end

local function GetHealth()

	return 1
end

	local hum = GetHumanoid()
	if hum then return hum.Health end
	return 100
end

-- Main Update Loop
local function Update(dt)
	local _curHealth = 1
	SafePcall(function() _curHealth = GetHealth() end)
	local function finish(s, l)
		UpdateLayerStatics(dt, s, l)
		UpdateCloseVignette(dt, s)
		local singleOn = UpdateEyeSingles(dt, s)
		UpdateEyes(s, l, singleOn)
		UpdateVisualizer(dt, s, l)
		SafePcall(function() UpdateDesperationAnims(s) end)
		SafePcall(function() UpdateBreath(dt, s, _curHealth) end)
	end
	
	GlobalRefreshT = GlobalRefreshT + (dt or 0.016)
	if GlobalRefreshT >= 5 then
		GlobalRefreshT = 0
		SafePcall(function() RefreshGlobalIds() end)
		if Sounds.idle then SafePcall(function() Sounds.idle.Volume = CONFIG.VOLUME_IDLE end) end
		if Sounds.heartbeat then SafePcall(function() Sounds.heartbeat.Volume = CONFIG.VOLUME_HEARTBEAT end) end
	end
	
	if not CONFIG.IS_ACTIVE then
		if Sounds.heartbeat then StopHeartbeat() end
		finish("idle", 0)
		return
	end
	
	if IsDead then
		if Sounds.heartbeat then StopHeartbeat() end
		StopAllSounds()
		finish("dead", 0)
		return
	end
	-- Spectator mode (v2.5p): mute chase while spectating others
	local _spec = false
	SafePcall(function() _spec = IsSpectating() end)
	if _spec then
		if CONFIG.SPECTATOR_MUTE then
			if Sounds.chase or Sounds.lastChance or Sounds.finalChance then StopAllSounds() end
			if Sounds.heartbeat then StopHeartbeat() end
		end
		finish("idle", 0)
		SafePcall(function()
			if CONFIG.SPECTATOR_DIM_EYES and CloseImg then CloseImg.Visible = false end
		end)
		return
	end
	
	local _, dist = GetCachedTarget()
	local health = GetHealth()
	local hpAbs = GetHealthAbs()
	local theme = GetCurrentTheme()
	if not theme then finish("idle", 0) return end
	local chaseDist = GetChaseDist(theme)
	
	-- Final Chance State
	if hpAbs <= CONFIG.FINAL_CHANCE_HP and hpAbs > 0 then
		if isChaseActive then
			if Sounds.chase then
				StopSound(Sounds.chase)
				Sounds.chase = nil
			end
			isChaseActive = false
		end
		if Sounds.lastChance then
			StopSound(Sounds.lastChance)
			Sounds.lastChance = nil
		end
		StopLastLayers()
		if not Sounds.finalChance and not theme.hasFinalCustom then
			CurrentFinalVariant = math.random(1, 2)
		end
		StopLayersAndIdle()
		SetFinalSound(theme)
		SetFinalFX(true)
		finish("final", 0)
		return
	else
		if Sounds.finalChance then
			StopSound(Sounds.finalChance)
			Sounds.finalChance = nil
		end
		SetFinalFX(false)
	end
	
	-- Low HP State
	if theme and theme.hasLastLayers then
		SafePcall(function() RefreshThemeAssets(theme) end)
	end
	
	local lowHP = health <= CONFIG.HEARTBEAT_HEALTH and health > 0
	SafePcall(function() UpdateHeartbeat(health, lowHP, true) end)
	
	if lowHP and theme.hasLastLayers and (HasLastLayerFiles(theme) or HasLastChanceFile(theme)) then
		local lastChaseRange = CONFIG.LAST_CHANCE_DISTANCE
		local lastLayerRange = CONFIG.LAST_CHANCE_LAYER_DISTANCES[1]
		
		if dist < lastChaseRange then
			if isChaseActive then
				if Sounds.chase then
					StopSound(Sounds.chase)
					Sounds.chase = nil
				end
				isChaseActive = false
			end
			StopNormalLayers()
			if Sounds.idle then
				StopSound(Sounds.idle)
				Sounds.idle = nil
			end
			StopLastLayers()
			if not Sounds.lastChance then
				CurrentLastChanceVariant = math.random(1, 2)
			end
			SetLastChanceSound(theme)
			finish("lastchance", 3)
			return
		elseif dist < lastLayerRange then
			if isChaseActive then
				if Sounds.chase then
					StopSound(Sounds.chase)
					Sounds.chase = nil
				end
				isChaseActive = false
			end
			if Sounds.lastChance then
				StopSound(Sounds.lastChance)
				Sounds.lastChance = nil
			end
			if Sounds.idle then
				StopSound(Sounds.idle)
				Sounds.idle = nil
			end
			StopNormalLayers()
			local ln = UpdateLastLayers(dist, theme)
			finish("lastlayer", ln)
			return
		end
	end
	
	-- Normal Chase Logic
	local inChase = dist < chaseDist
	local inLC = dist < CONFIG.LAST_CHANCE_DISTANCE
	local useLC = lowHP and inLC and theme.hasLastChance and HasLastChanceFile(theme) and not theme.hasLastLayers
	
	if useLC then
		if isChaseActive then
			if Sounds.chase then
				StopSound(Sounds.chase)
				Sounds.chase = nil
			end
			isChaseActive = false
		end
		if not Sounds.lastChance then
			CurrentLastChanceVariant = math.random(1, 2)
		end
		StopLayersAndIdle()
		StopLastLayers()
		SetLastChanceSound(theme)
		finish("lastchance", 3)
	else
		if Sounds.lastChance then
			StopSound(Sounds.lastChance)
			Sounds.lastChance = nil
		end
		StopLastLayers()
		
		if inChase then
			chaseExitTimer = 0
			if not isChaseActive then
				StopLayersAndIdle()
				isChaseActive = true
			end
			local inten = 1 - Clamp(SafeDiv(dist, (chaseDist ~= 0 and chaseDist or 45), 0), 0, 1)
			SetChaseSound(theme, CONFIG.VOLUME_CHASE * (0.5 + inten * 0.5))
			finish("chase", 0)
		else
			if isChaseActive then
				chaseExitTimer = chaseExitTimer + (dt or 0.016)
				if chaseExitTimer >= CONFIG.CHASE_EXIT_DELAY then
					if Sounds.chase then
						StopSound(Sounds.chase)
						Sounds.chase = nil
					end
					isChaseActive = false
					chaseExitTimer = 0
				end
			end
			
			if not isChaseActive then
				if theme.hasLayers and dist < CONFIG.LAYER_DISTANCES[1] then
					if Sounds.idle then
						StopSound(Sounds.idle)
						Sounds.idle = nil
					end
					local ln = UpdateLayers(dist, theme)
					finish("layer", ln)
				elseif GetIdleId(theme) ~= "" then
					StopLayersAndIdle()
					SetIdleSound(theme)
					finish("idle", 0)
				else
					StopLayersAndIdle()
					if Sounds.chase then StopSound(Sounds.chase); Sounds.chase = nil end
					if Sounds.lastChance then StopSound(Sounds.lastChance); Sounds.lastChance = nil end
					if Sounds.finalChance then StopSound(Sounds.finalChance); Sounds.finalChance = nil end
					StopLastLayers()
					isChaseActive = false
					chaseExitTimer = 0
					finish("silence", 0)
				end
			else
				finish("chase", 0)
			end
		end
	end
end

-- Smooth theme transitions (v2.5p)
local function SmoothSwitchTheme(targetIdx)
  targetIdx = Clamp(math.floor(tonumber(targetIdx) or 1), 1, #THEMES)
  if ThemeFading then return end
  if targetIdx == (tonumber(CONFIG.CURRENT_THEME) or 1) and CONFIG.IS_ACTIVE then return end
  ThemeFading = true
  local dur = Clamp(tonumber(CONFIG.THEME_FADE_DURATION) or 0.8, 0.1, 3)
  -- fade live volumes out instead of hard cut
  SafePcall(function()
    for _, s in pairs({Sounds.chase, Sounds.idle, Sounds.heartbeat, Sounds.lastChance, Sounds.finalChance}) do
      if s and s.Parent then
        TweenService:Create(s, TweenInfo.new(dur * 0.7), {Volume = 0}):Play()
      end
    end
    for i = 1, 3 do
      if Sounds.layers[i] and Sounds.layers[i].Parent then
        TweenService:Create(Sounds.layers[i], TweenInfo.new(dur * 0.7), {Volume = 0}):Play()
      end
      if Sounds.lastLayers[i] and Sounds.lastLayers[i].Parent then
        TweenService:Create(Sounds.lastLayers[i], TweenInfo.new(dur * 0.7), {Volume = 0}):Play()
      end
    end
  end)
  task.delay(dur, function()
    SafePcall(function()
      StopAllSounds()
      CONFIG.IS_ACTIVE = false
      CONFIG.CURRENT_THEME = targetIdx
      LastThemeIdx = targetIdx
      local th = THEMES[targetIdx]
      if th then PreloadIds(CollectThemeIds(th)) end
      SaveConfig(true)
    end)
    ThemeFading = false
  end)
end

-- GUI Construction
local function EnsureSharedDrag()
	if SharedDrag.conn then return end
	SharedDrag.conn = TrackConn(UserInputService.InputChanged:Connect(function(input)
		if SharedDrag.active and SharedDrag.gui and SharedDrag.start and SharedDrag.origin then
			if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
				local delta = input.Position - SharedDrag.start
				pcall(function()
					SharedDrag.gui.Position = UDim2.new(SharedDrag.origin.X.Scale, SharedDrag.origin.X.Offset + delta.X,
						SharedDrag.origin.Y.Scale, SharedDrag.origin.Y.Offset + delta.Y)
				end)
			end
		end
	end))
	SharedDrag.endConn = TrackConn(UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			SharedDrag.active = false
		end
	end))
end
local function MakeDraggable(main, handle)
	EnsureSharedDrag()
	handle.Active = true
	TrackConn(handle.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			SharedDrag.active = true
			SharedDrag.gui = main
			SharedDrag.start = input.Position
			SharedDrag.origin = main.Position
		end
	end))
	TrackConn(handle.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			SharedDrag.active = false
		end
	end))
end

-- Character Management
local function OnCharAdded(char)
	local hum = char:WaitForChild("Humanoid", 5)
	if not hum then return end
	
	IsDead = false
	SetFinalFX(false)
	IsCrouching = false
	IsSpeeding = false
	SpeedAutoBroken = false
	SpeedBreakBelowTime = 0
	WasSpeedBeforeCrouch = false
	StopCrouchTracks()
	SafePcall(function() StopDesperationTracks() end)
	
	OrigWalkSpeed = hum.WalkSpeed
	NormalWalkSpeed = hum.WalkSpeed
	AccelCur = nil
	LastUncrouchAt = 0
	AdrenActiveUntil, AdrenCooldownUntil = 0, 0
	if AdrenAnimTrack then pcall(function() AdrenAnimTrack:Stop(0.2) end) AdrenAnimTrack = nil end
	BackwardBlend = 0
	SafePcall(function() EnsureFootEmitter(char) end)
	SafePcall(function() RefreshAdrenBtn() end)
	-- SetSpawn: play respawn anim + optimized teleport (1 teleport, verify, retry up to N)
	if SpawnArmed and SavedSpawnCF then
		local _spawnCF = SavedSpawnCF
		-- 1) Respawn animation (once the new character's Animator is ready)
		task.spawn(function()
			SafePcall(function()
				local hum = char:WaitForChild("Humanoid", 5)
				if hum then hum:WaitForChild("Animator", 3) end
				PlaySpawnAnim(char, CONFIG.RESPAWN_ANIM_ID)
			end)
		end)
		-- 2) Teleport: single call, verify arrival, retry up to N attempts (1s apart)
		task.spawn(function()
			local maxAttempts = math.max(1, math.floor(tonumber(CONFIG.RESPAWN_TELEPORT_ATTEMPTS) or 3))
			local retryDelay = tonumber(CONFIG.RESPAWN_TELEPORT_DELAY) or 1
			if retryDelay < 0.1 then retryDelay = 0.1 end
			local arriveDist = tonumber(CONFIG.RESPAWN_TELEPORT_DIST) or 12
			local target = _spawnCF + Vector3.new(0, 3, 0)
			local attempts = 0
			while attempts < maxAttempts do
				if not SpawnArmed then break end
				attempts = attempts + 1
				local h = char:FindFirstChild("HumanoidRootPart")
				if h then
					pcall(function()
						h.CFrame = target
						h.AssemblyLinearVelocity = Vector3.zero
						h.AssemblyAngularVelocity = Vector3.zero
					end)
				end
				task.wait(retryDelay)
				local h2 = char:FindFirstChild("HumanoidRootPart")
				if h2 then
					local d = (h2.Position - _spawnCF.Position).Magnitude
					if d <= arriveDist then break end
				end
			end
			-- cleanup after attempts complete
			SpawnArmed = false
			SpawnUsed = false
			SpawnTransparentActive = false
			if SpawnTransparentHook then pcall(function() SpawnTransparentHook:Disconnect() end) SpawnTransparentHook = nil end
			pcall(function() RefreshSpawnBtn() end)
		end)
	else
		SpawnTransparentActive = false
	end
	
	task.delay(0.35, function()
		SafePcall(function() PlayRespawnSfx() end)
	end)
	
	local pg = LocalPlayer:FindFirstChild("PlayerGui")
	if pg and pg:FindFirstChild("ChaseGUI_Dead") then
		pg.ChaseGUI_Dead:Destroy()
	end
	
	if DiedConn then pcall(function() DiedConn:Disconnect() end) DiedConn = nil end
	DiedConn = TrackConn(hum.Died:Connect(function()
		IsDead = true
		if IsCrouching then
			IsCrouching = false
			StopCrouchTracks()
		end
		AdrenActiveUntil = 0
		if AdrenAnimTrack then pcall(function() AdrenAnimTrack:Stop(0.2) end) AdrenAnimTrack = nil end
		SafePcall(function() StopDesperationTracks() end)
		SafePcall(function() UpdateFootsteps(0, false, false) end)
		StopAllSounds()
		SafePcall(function() ApplyDucking(false) end)
		BuildGUI(true)
	end))
	
	task.delay(0.2, function()
		RefreshCrouchBtn()
		RefreshSpeedBtn()
		SafePcall(function() RefreshRadarBtn() end)
		SafePcall(function() RefreshSpawnBtn() end)
		SafePcall(function() RefreshAdrenBtn() end)
		SafePcall(function() RefreshTpSpawnBtn() end)
		SafePcall(function() EnsureFootEmitter(char) end)
	end)
end

		if RespawnLastSnd then
			RespawnLastSnd:Stop()
			RespawnLastSnd:Destroy()
			RespawnLastSnd = nil
		end
	end)
	
	local ok, s = pcall(function()
		local snd = Instance.new("Sound")
		snd.Name = "RespawnSfx"
		snd.SoundId = assetId
		snd.Volume = CONFIG.RESPAWN_VOLUME or 1
		snd.Looped = false
		snd.PlayOnRemove = false
		snd.Parent = game:GetService("SoundService")
		return snd
	end)
	
	if ok and s then
		RespawnLastSnd = s
		pcall(function()
			pcall(function()
				if not s.IsLoaded then s.Loaded:Wait(5) end
			end)
			s:Play()
			print("[RespawnSfx] playing " .. tostring(s.SoundId))
		end)
		task.delay(8, function()
			pcall(function()
				if RespawnLastSnd == s then RespawnLastSnd = nil end
				s:Destroy()
			end)
		end)
	end
end

PlayRespawnSfx = function()
	if not CONFIG.RESPAWN_ENABLED then return end
	
	local f1 = CONFIG.RESPAWN_FILE_1
	local f2 = CONFIG.RESPAWN_FILE_2
	local a1 = RespawnAsset(f1)
	local a2 = RespawnAsset(f2)
	
	if (not a1 or a1 == "") and (not a2 or a2 == "") then
		warn("[RespawnSfx] both missing in " .. ASSET_FOLDER)
		return
	end
	
	local mode = string.lower(tostring(CONFIG.RESPAWN_MODE or "random"))
	
	if mode == "both" then
		if a1 and a1 ~= "" then RespawnPlayId(a1) end
		if a2 and a2 ~= "" then
			task.delay(0.05, function()
				local ok, s = pcall(function()
					local snd = Instance.new("Sound")
					snd.Name = "RespawnSfx2"
					snd.SoundId = a2
					snd.Volume = CONFIG.RESPAWN_VOLUME or 1
					snd.Looped = false
					snd.Parent = game:GetService("SoundService")
					return snd
				end)
				if ok and s then
					pcall(function() s:Play() end)
					task.delay(8, function() pcall(function() s:Destroy() end) end)
				end
			end)
		end
		return
	end
	
	if mode == "alternate" then
		RespawnAltIdx = (RespawnAltIdx % 2) + 1
		local pick = (RespawnAltIdx == 1) and a1 or a2
		if not pick or pick == "" then pick = (a1 and a1 ~= "") and a1 or a2 end
		RespawnPlayId(pick)
		return
	end
	
	-- random mode
	local use1 = (math.random(1, 2) == 1)
	local pick = use1 and a1 or a2
	if not pick or pick == "" then pick = (a1 and a1 ~= "") and a1 or a2 end
	RespawnPlayId(pick)
end

-- ================= v3.0 ANIMATION REPLACER =================
-- Custom locomotion set (Idle/Walk/Run/Jump/Fall), merged from animation_replacer_v2.
-- Yields to crouch + desperation. Toggle: CONFIG.ANIM_REPLACER_ENABLED.
-- Perf: tracks are preloaded once per character; foreign-track cleanup is throttled.
local function StartAnimReplacer()
	local ARPlayers = game:GetService("Players")
	local ARRun = game:GetService("RunService")
	local ARLocal = ARPlayers.LocalPlayer
	if not ARLocal then return end

	pcall(function()
		if _G.__AnimReplacerCleanup and typeof(_G.__AnimReplacerCleanup) == "function" then
			_G.__AnimReplacerCleanup()
		end
	end)

	local DEFAULT_IDS = {
		Idle = "rbxassetid://71280785180715",
		Walk = "rbxassetid://100258502351639",
		Run  = "rbxassetid://94027322797812",
		Jump = "rbxassetid://125772394585171",
		Fall = "rbxassetid://113767290702894",
	}
	local ORDER = {"Idle", "Walk", "Run", "Jump", "Fall"}
	local SCAN_INTERVAL = 0.25
	local conns = {}
	local tracks = {}
	local enabled = (CONFIG.ANIM_REPLACER_ENABLED ~= false)
	local current = nil
	local yielded = false
	local scanAccum = 0

	local function ids()
		local t = CONFIG.ANIM_REPLACER_IDS
		if typeof(t) == "table" and next(t) ~= nil then return t end
		return DEFAULT_IDS
	end

	local function getAnimator()
		local char = ARLocal.Character
		if not char then return nil, nil end
		local hum = char:FindFirstChildOfClass("Humanoid")
		if not hum then return nil, nil end
		local animator = hum:FindFirstChildOfClass("Animator")
		return animator, hum
	end

	local function stopAll()
		for _, name in ipairs(ORDER) do
			local tr = tracks[name]
			if tr then pcall(function() if tr.IsPlaying then tr:Stop(0.15) end end) end
		end
		current = nil
	end

	local function loadTracks(animator)
		stopAll()
		tracks = {}
		local t = ids()
		for _, name in ipairs(ORDER) do
			local id = t[name]
			if id and id ~= "" then
				local anim = Instance.new("Animation")
				anim.AnimationId = id
				local ok, track = pcall(function() return animator:LoadAnimation(anim) end)
				if ok and track then
					pcall(function() track.Name = name end)
					track.Priority = Enum.AnimationPriority.Action
					track.Looped = (name ~= "Jump")
					tracks[name] = track
				end
			end
		end
	end

	local function playState(name, speed)
		local tr = tracks[name]
		if not tr then return end
		-- Reverse (backward) support: mirror the main backward system using the SAME
		-- smoothed condition (BackwardBlend > 0.5) so the replacer and
		-- UpdateBackwardAnim() never fight over AnimationTrack.Speed. Jump/Fall are
		-- never reversed, matching the main system.
		local dir = 1
		if name ~= "Jump" and name ~= "Fall" and CONFIG.BACKWARD_ANIM_ENABLED and (BackwardBlend or 0) > 0.5 then
			dir = -(tonumber(CONFIG.BACKWARD_ANIM_SPEED) or 1)
		end
		local spd = (speed or 1) * dir
		if current == name then
			if tr.IsPlaying then
				tr:AdjustSpeed(spd)
			else
				tr:Play(0.15, 1, spd)
			end
			return
		end
		if current and tracks[current] then pcall(function() tracks[current]:Stop(0.15) end) end
		tr:Play(0.15, 1, spd)
		current = name
	end

	local function hookCharacter(char)
		stopAll()
		tracks = {}
		yielded = false
		scanAccum = 0
		task.spawn(function()
			local hum = char:WaitForChild("Humanoid", 3)
			local animator = hum and hum:WaitForChild("Animator", 3)
			if not animator then return end
			loadTracks(animator)
			if next(tracks) ~= nil then
				local animate = char:FindFirstChild("Animate")
				if animate then animate.Disabled = true end
			end
		end)
	end

	table.insert(conns, ARLocal.CharacterAdded:Connect(hookCharacter))
	if ARLocal.Character then hookCharacter(ARLocal.Character) end

	table.insert(conns, ARRun.RenderStepped:Connect(function(dt)
		if not enabled then
			if not yielded then stopAll() yielded = true end
			return
		end
		local animator, hum = getAnimator()
		if not animator or not hum or hum.Health <= 0 then return end

		if IsCrouching == true or _G.__DesperationActive == true then
			if not yielded then stopAll() yielded = true end
			return
		end
		yielded = false

		scanAccum = scanAccum + (dt or 0.016)
		if scanAccum >= SCAN_INTERVAL then
			scanAccum = 0
			if not CONFIG.PERF_MODE then
				for _, t in ipairs(animator:GetPlayingAnimationTracks()) do
					local isOurs = false
					for _, name in ipairs(ORDER) do if tracks[name] == t then isOurs = true break end end
					if not isOurs and t.Priority.Value <= Enum.AnimationPriority.Movement.Value then
						pcall(function() t:Stop(0.15) end)
					end
				end
			end
		end

		local vel = hum.RootPart and hum.RootPart.AssemblyLinearVelocity or Vector3.zero
		local hSpeed = Vector3.new(vel.X, 0, vel.Z).Magnitude
		local moving = hum.MoveDirection.Magnitude > 0.1
		local runMin = tonumber(CONFIG.ANIM_REPLACER_RUN_MIN_SPEED) or 18
		if runMin < 1 then runMin = 1 end

		if hum.FloorMaterial == Enum.Material.Air then
			if vel.Y > 1 then
				playState("Jump", 1)
			else
				playState("Fall", 1)
			end
		elseif not moving or hSpeed < 1 then
			playState("Idle", 1)
		elseif hSpeed >= runMin then
			playState("Run", hSpeed / runMin)
		else
			playState("Walk", math.clamp(hSpeed / 12, 0.8, 1.4))
		end
	end))

	_G.__AnimReplacerCleanup = function()
		for _, c in ipairs(conns) do pcall(function() c:Disconnect() end) end
		stopAll()
	end

	_G.__AnimReplacerSetEnabled = function(v)
		enabled = (v ~= false)
		if not enabled then stopAll() yielded = true else yielded = false end
	end

	print("[AnimReplacer v3] loaded")
end

SafePcall(function() StartAnimReplacer() end)

print("[Chase] v3.0 loaded. Test: _G.RespawnSfx.Play() | Cleanup: _G.ChaseCleanup()")

-- ================= v2.5p AUTO-CLEANUP =================
local _cleaned = false
local function ChaseCleanup(reason)
  if _cleaned then return end
  _cleaned = true
  pcall(function() print("[Chase] Cleanup (" .. tostring(reason or "destroy") .. ")") end)
  for _, c in ipairs(AllConns) do pcall(function() c:Disconnect() end) end
  if DiedConn then pcall(function() DiedConn:Disconnect() end) end
  pcall(function() RunService:UnbindFromRenderStep("ChaseBackwardFix") end)
  SafePcall(function() if _G.__AnimReplacerCleanup then _G.__AnimReplacerCleanup() end end)
  SafePcall(function()
    if IsCrouching then IsCrouching = false end
    if IsSpeeding then IsSpeeding = false end
    AdrenActiveUntil = 0
    if AdrenAnimTrack then pcall(function() AdrenAnimTrack:Stop(0.2) end) AdrenAnimTrack = nil end
    StopCrouchTracks()
    pcall(function() StopDesperationTracks() end)
  end)
  SafePcall(function()
    -- instant destroy (no fade) for cleanup path
    for i = 1, 3 do
      if Sounds.layers[i] then pcall(function() Sounds.layers[i]:Destroy() end) Sounds.layers[i] = nil end
      if Sounds.lastLayers[i] then pcall(function() Sounds.lastLayers[i]:Destroy() end) Sounds.lastLayers[i] = nil end
    end
    for _, k in ipairs({"chase","idle","heartbeat","lastChance","finalChance"}) do
      if Sounds[k] then pcall(function() Sounds[k]:Destroy() end) Sounds[k] = nil end
    end
    for _, s in ipairs(ActiveSounds) do pcall(function() if s and s.Parent then s:Destroy() end end) end
    if RespawnLastSnd then pcall(function() RespawnLastSnd:Destroy() end) end
  end)
  SafePcall(function()
    for _, h in pairs(RadarHighlights or {}) do pcall(function() if h then h.FillTransparency = 1 h.OutlineTransparency = 1 end end) end
  end)
  SafePcall(function() SetFinalFX(false) end)
  SafePcall(function()
    local pg = LocalPlayer:FindFirstChildOfClass("PlayerGui") or LocalPlayer:FindFirstChild("PlayerGui")
    if pg then
      for _, n in ipairs({"ChaseGUI","ChaseGUI_Dead","ChaseToggleGui","SpeedBtnGui","CrouchBtnGui","RadarBtnGui","SpawnBtnGui","AdrenBtnGui","TpSpawnBtnGui","TpModeBtnGui","DirCueGui","BreathFX","TransFadeGui","RespawnSfxGui","CloseDeathFX","ChaseEyes","ChaseEyeSingles","EyeSingles","ChaseStatics","ChaseVisualizer"}) do
        local g = pg:FindFirstChild(n)
        if g then pcall(function() g:Destroy() end) end
      end
    end
    if FinalCC then pcall(function() FinalCC:Destroy() end) FinalCC = nil end
  end)
  table.clear(AllConns)
end
_G.ChaseCleanup = ChaseCleanup
_G.__ChaseCleanup = ChaseCleanup

end))

  end)
